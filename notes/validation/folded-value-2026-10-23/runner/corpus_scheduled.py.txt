from pathlib import Path
import json,os,re,signal,subprocess,time
root=Path.cwd()
out=root/'.agent-files/folded-value-validation/corpus'
rows=json.loads((out/'results.json').read_text())
# Preserve the actual batch exit and completed siblings before replacing only
# our paused scheduler. The failed dependency has already exceeded 600 seconds.
if 'evalCompilerBytecodeAbiInternalCallsTheory' not in rows:
    child=923099
    stat=Path(f'/proc/{child}/stat').read_text().rsplit(')',1)[1].split()
    assert stat[0]=='Z'
    batch_exit=int(stat[49])>>8
    driver=662981
    assert Path(f'/proc/{driver}/cmdline').read_bytes().replace(b'\0',b' ').decode().strip()=='python3 -I .agent-files/folded-value-validation/corpus_parallel.py'
    os.kill(driver,signal.SIGKILL)
    text=(out/'batch-01.log').read_text()
    for target in ['evalCompilerBytecodeAbiExternalDynamicTheory','evalCompilerBytecodeAbiEventsTheory','evalCompilerBytecodeAbiInternalCallsTheory']:
        if re.search(r'^'+target+r' built$',text,re.M):
            rows[target]={'exit':0,'batch_exit':batch_exit,'log':'batch-01.log','built_message_observed':True}
    rows['evalCompilerBytecodeAbiInternalCallsTheory']={'exit':124,'batch_exit':batch_exit,'status':'DEPENDENCY_EVALUATION_TIMEOUT','dependency':'evalCompilerBytecodeAbiInternalTupleLoweringTheory','log':'batch-01.log','reason':'Terminated only its HOL evaluation after more than 600 seconds without a completed step. Top-level ML EVAL does not obey tactic timeout.'}
    (out/'results.json').write_text(json.dumps(rows,indent=2)+'\n')
wrapper=(root/'lowering/defs/bytecode/python-o1-bytecode-fixtures').read_text()
targets=list(dict.fromkeys(re.findall(r'\bevalCompilerBytecode\w*Theory\b',wrapper)))
pending=[t for t in targets if t not in rows]
# Identify which roots actually depend on a failing theory; cancelled siblings
# get another attempt, while known failed dependencies are not retried.
deps={}
for file in subprocess.check_output(['git','ls-files','*Script.sml'],text=True).splitlines():
    p=root/file
    content=p.read_text()
    name=re.search(r'^Theory\s+(\w+)',content,re.M)
    ancestors=re.search(r'\bAncestors\s+(.+?)(?:\bLibs\b|\n\s*\n)',content,re.S)
    if name:deps[name[1]+'Theory']=[v+'Theory' for v in re.findall(r'\b[A-Za-z_]\w*\b',ancestors[1])] if ancestors else []
def closure(name,seen=None):
    if seen is None:seen=set()
    if name in seen:return seen
    seen.add(name)
    for dep in deps.get(name,[]):closure(dep,seen)
    return seen
def active_hol(parent):
    records=[]
    for line in subprocess.check_output(['ps','-eo','pid,ppid,comm'],text=True).splitlines()[1:]:
        pid,ppid,comm=line.split(None,2);records.append((int(pid),int(ppid),comm))
    descendants={parent}
    for _ in range(8):
        for pid,ppid,comm in records:
            if ppid in descendants:descendants.add(pid)
    for pid,ppid,comm in records:
        if pid in descendants and comm=='hol':
            try:
                cwd=Path(f'/proc/{pid}/cwd').resolve()
                if str(cwd).startswith(str(root/'.holbuild/stage')):yield pid,cwd
            except OSError:pass
attempts={}
batch_index=100
while pending:
    batch=pending;pending=[]
    command=['holbuild','--maxheap','12000','-j3','build','--force=theory','--tactic-timeout','600',*batch]
    logpath=out/f'bounded-{batch_index:02}.log';batch_index+=1
    stopped=[]
    start=time.time()
    with logpath.open('w') as log:
        process=subprocess.Popen(command,stdout=log,stderr=subprocess.STDOUT)
        while process.poll() is None:
            for pid,cwd in active_hol(process.pid):
                path=cwd/'holbuild-build.log'
                if path.exists() and time.time()-path.stat().st_mtime>600:
                    native=path.read_text()
                    name=re.search(r'Created theory "([^"]+)"',native)
                    entry={'pid':pid,'theory':name[1]+'Theory' if name else 'unknown','stage':str(cwd),'reason':'No completed evaluation step for over 600 seconds'}
                    stopped.append(entry)
                    try:os.kill(pid,signal.SIGTERM)
                    except ProcessLookupError:pass
                    (out/'timeouts.json').write_text(json.dumps(stopped,indent=2)+'\n')
            time.sleep(2)
    text=logpath.read_text()
    failures=re.findall(r'/([^/\s]+Theory)/instrumented-failure\.log',text)
    failures+= [entry['theory'] for entry in stopped]
    for target in batch:
        attempts[target]=attempts.get(target,0)+1
        built=bool(re.search(r'^'+re.escape(target)+r' built$',text,re.M))
        failed_deps=sorted(set(failures)&closure(target))
        if process.returncode==0 or built:
            rows[target]={'exit':0,'argv':command,'log':logpath.name,'built_message_observed':built,'seconds':round(time.time()-start,3)}
        elif failed_deps or attempts[target]>=2:
            rows[target]={'exit':124 if stopped else process.returncode,'batch_exit':process.returncode,'status':'DEPENDENCY_EVALUATION_TIMEOUT' if stopped else 'BUILD_FAILURE','dependencies':failed_deps,'argv':command,'log':logpath.name,'seconds':round(time.time()-start,3),'stopped':stopped}
        else:
            pending.append(target)
            print('REQUEUE CANCELLED SIBLING',target,flush=True)
        if target in rows:print(target,json.dumps(rows[target]),flush=True)
    (out/'results.json').write_text(json.dumps(rows,indent=2)+'\n')
print('ALL REQUESTED TARGETS ATTEMPTED',flush=True)
